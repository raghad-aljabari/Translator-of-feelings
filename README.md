## كيف يعمل النظام

**المرحلة الأولى — الالتقاط**
كاميرا تراقب وجه الطفل بشكل مستمر. صُممت لتعمل في الخلفية دون أن يلاحظها الطفل أو يتأثر بها.

**المرحلة الرابعة — الإشعار**
يصل الأهل إشعاراً فورياً بواسطة تطبيق يعرض الحالة الشعورية لطفلهم بطريقة مرئية وبسيطة — سعادة أو توتر أو ألم.

## تشغيل المشروع

**المتطلبات**

- LEGO SPIKE Prime
- laptop camera (or any camera kind)
- MicroPython مثبّت على الوحدة

**خطوات التشغيل**

```
1.  قم بتوصيل وحدة *SPIKE Prime* بالكمبيوتر عبر الـ Bluetooth أو كابل USB.
2.  تأكد من رفع كود الـ *MicroPython* الخاص بالحركات على وحدة الـ Hub.
3.  قم بتشغيل البرنامج على اللابتوب لتفعيل *كاميرا الجهاز* والبدء بالذكاء الاصطناعي.
4.  سيبدأ النظام بتحليل المشاعر في الوقت الفعلي وتوجيه الروبوت للتفاعل بناءً عليها```
5.يتم ارسال اشعار للاهل
**الكود الأساسي**

```python
import hub
import sys
import time

# يستقبل المشاعر من الكاميرا ويتفاعل

print(" مترجم الاحاسيس جاهز!")
print(" في انتظار البيانات...")

last_emotion = ""

while True:
    try:
        data = sys.stdin.readline().strip()

        # بس يتفاعل إذا تغير الإحساس
        if data and data != last_emotion:
            last_emotion = data

            if data == "joy":
                hub.light_matrix.show_image(hub.light_matrix.IMAGE_HAPPY)
                hub.sound.beep(784, 400)
                print("😊 المشاعر: سعيد")

            elif data == "sad":
                hub.light_matrix.show_image(hub.light_matrix.IMAGE_SAD)
                hub.sound.beep(330, 600)
                print("😢 المشاعر: حزين")

            elif data == "angry":
                hub.light_matrix.show_image(hub.light_matrix.IMAGE_ANGRY)
                hub.sound.beep(200, 200)
                time.sleep(0.1)
                hub.sound.beep(200, 200)
                time.sleep(0.1)
                hub.sound.beep(200, 200)
                print("😠 المشاعر: غاضب")

            elif data == "fear":
                hub.light_matrix.show_image(hub.light_matrix.IMAGE_GHOST)
                hub.sound.beep(600, 100)
                time.sleep(0.1)
                hub.sound.beep(500, 100)
                time.sleep(0.1)
                hub.sound.beep(400, 100)
                print("😨 المشاعر: خايف")

            elif data == "subrised":
                hub.light_matrix.show_image(hub.light_matrix.IMAGE_SURPRISED)
                hub.sound.beep(1047, 300)
                time.sleep(0.1)
                hub.sound.beep(1200, 500)
                print("😮 المشاعر: متفاجئ")

            elif data == "norm":
                hub.light_matrix.show_image(hub.light_matrix.IMAGE_MEH)
                hub.sound.beep(440, 300)
                print("😐 المشاعر: عادي")

            else:
                print("⚠️ مشاعر غير معروفة: " + data)

    except Exception as e:
        print("❌ خطأ: " + str(e))
```

  emojiEl.classList.add('pop');

  document.getElementById('cur-emotion').textContent = emotion;
  document.getElementById('cur-time').textContent = 'آخر رصد: ' + time;
  document.getElementById('cur-conf').textContent = 'ثقة: ' + confidence + '%';

  // تحديث مؤشر الاتصال
  const bar = document.getElementById('conn-bar');
  bar.className = 'conn-bar conn-on';
  document.getElementById('conn-text').textContent = '● متصل — آخر تحديث: ' + time;

  // إضافة للسجل المحلي
  const entry = { emotion, confidence, time, date, timestamp: data.timestamp };
  moodLog.unshift(entry);
  if (moodLog.length > 200) moodLog = moodLog.slice(0, 200); // نحتفظ بأخر 200
  localStorage.setItem('moodLog_v2', JSON.stringify(moodLog));

  renderLog();
  updateStats();
  checkEmergency(key, emotion);

  // إشعار للهاتف
  if (Notification.permission === 'granted') {
    new Notification('🧠 مترجم الأحاسيس', {
      body: 'طفلك يشعر بـ' + (arMap[key]||emotion) + ' — ثقة: ' + confidence + '%',
      icon: '/icon-192.png',
      tag: 'emotion'
    });
  }
});

function checkEmergency(key, emotion) {
  if (negatives.includes(key)) {
    if (!negStart) negStart = Date.now();
    const mins = (Date.now() - negStart) / 60000;
    negStreak++;
    if (mins >= 2) {
      document.getElementById('emergency').classList.add('show');
      document.getElementById('em-title').textContent = '🚨 شعور مستمر!';
      document.getElementById('em-sub').textContent = 'طفلك يشعر بـ"' + (arMap[key]||emotion) + '" منذ أكثر من دقيقتين';
      if (Notification.permission === 'granted') {
        new Notification('🚨 تنبيه طوارئ', {
          body: 'طفلك يشعر بـ"' + (arMap[key]||emotion) + '" منذ أكثر من دقيقتين!',
          requireInteraction: true,
          tag: 'emergency'
        });
      }
    }
  } else {
    negStart = null; negStreak = 0;
    document.getElementById('emergency').classList.remove('show');
  }
}

function renderLog() {
  const list = document.getElementById('log-list');
  document.getElementById('log-count').textContent = moodLog.length + ' سجل';
  if (moodLog.length === 0) {
    list.innerHTML = `<div class="empty-state"><span class="icon">📋</span>لم يُرصد أي مشاعر بعد</div>`;
    return;
  }
  list.innerHTML = moodLog.slice(0, 30).map(e => {
    const k = e.emotion.toLowerCase();
    const urg = negatives.includes(k);
    return `<div class="log-item ${urg?'urgent':''}">
      <div class="log-ew ${urg?'urgent':''}">${emojiMap[k]||'🤔'}</div>
      <div class="log-info">
        <div class="log-emotion">${e.emotion}</div>
        <div class="log-emotion-ar">${arMap[k]||e.emotion}</div>
        <div class="log-time">🕐 ${e.time} — ${e.date}</div>
      </div>
      <div class="log-conf ${urg?'urgent':''}">${e.confidence}%</div>
    </div>`;
  }).join('');
}

function updateStats() {
  let pos=0,neu=0,neg=0;
  moodLog.forEach(e=>{
    const k=e.emotion.toLowerCase();
    if(positives.includes(k)) pos++;
    else if(negatives.includes(k)) neg++;
    else neu++;
  });
  document.getElementById('s-pos').textContent=pos;
  document.getElementById('s-neu').textContent=neu;
  document.getElementById('s-neg').textContent=neg;
}
